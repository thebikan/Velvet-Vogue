-- Inventory reconciliation trail; amounts positive, direction encoded by delta sign.
CREATE TABLE IF NOT EXISTS inventory_movements (
  id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  variant_id BIGINT UNSIGNED NOT NULL,
  order_id BIGINT UNSIGNED NULL,
  actor_user_id BIGINT UNSIGNED NULL,
  change_quantity INT NOT NULL,
  reason ENUM('restock','sale','return','correction','damage') NOT NULL,
  note VARCHAR(255) NULL,
  created_at DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  KEY idx_stock_movements_variant(variant_id,created_at),
  KEY idx_stock_movements_order(order_id),

    REFERENCES orders(id) ON DELETE RESTRICT,

-- Durable asynchronous work queue. Must be claimed/processed by PHP CLI workers.
-- Limit types and validate payloads; never place credentials inside payload_json.
CREATE TABLE IF NOT EXISTS background_jobs (
  id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  job_type VARCHAR(80) NOT NULL,
  dedupe_key VARCHAR(160) NULL,
  payload_json JSON NOT NULL,
  status ENUM('pending','running','completed','failed') NOT NULL DEFAULT 'pending',
  attempts SMALLINT UNSIGNED NOT NULL DEFAULT 0,
  max_attempts SMALLINT UNSIGNED NOT NULL DEFAULT 5,
  available_at DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  locked_until DATETIME(6) NULL,
  last_error VARCHAR(1000) NULL,
  created_at DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  updated_at DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6)
    ON UPDATE CURRENT_TIMESTAMP(6),
  UNIQUE KEY uq_jobs_dedupe(job_type,dedupe_key),
  KEY idx_jobs_pickup(status,available_at,id),
  KEY idx_jobs_recovery(status,locked_until,id)
